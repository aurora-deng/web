cmake_minimum_required(VERSION 3.20)
project(nebula_companion LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This companion uses Windows Schannel, SQLite and DPAPI. Build the server from the repository root on Linux.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
set(NEBULA_BOOST_ROOT "" CACHE PATH "Explicitly approved Boost 1.92.0 headers; no downloads")
find_path(NEBULA_BOOST_INCLUDE boost/beast.hpp HINTS "${NEBULA_BOOST_ROOT}" NO_DEFAULT_PATH)
if(NOT NEBULA_BOOST_INCLUDE)
    message(FATAL_ERROR "Point NEBULA_BOOST_ROOT to the approved Boost headers. CMake never downloads dependencies.")
endif()
add_library(companion_json companion/JsonRuntime/JsonRuntime.cpp)
target_include_directories(companion_json PUBLIC "${NEBULA_BOOST_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(companion_json PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 BOOST_ASIO_NO_DEPRECATED)
target_link_libraries(companion_json PUBLIC Threads::Threads ws2_32 mswsock)
add_library(companion_storage storage/SqliteStore/SqliteStore.cpp storage/WindowsProtection/WindowsProtection.cpp)
target_link_libraries(companion_storage PUBLIC companion_json crypt32 bcrypt advapi32)
add_library(companion_private_data assistant/LocalKnowledge/LocalKnowledge.cpp assistant/PrivateHistory/PrivateHistory.cpp)
target_sources(companion_private_data PRIVATE assistant/DocumentImport/DocumentImport.cpp)
target_sources(companion_private_data PRIVATE assistant/ActionService/ActionService.cpp)
target_link_libraries(companion_private_data PUBLIC companion_storage)
add_library(companion_tls companion/SChannelStream/SChannelStream.cpp)
target_sources(companion_tls PRIVATE companion/BoundedPool/BoundedPool.cpp)
target_link_libraries(companion_tls PUBLIC companion_storage secur32)
add_library(companion_upstream companion/UpstreamTransport/UpstreamTransport.cpp)
target_sources(companion_upstream PRIVATE companion/HttpConnectionPool/HttpConnectionPool.cpp)
target_link_libraries(companion_upstream PUBLIC companion_tls)
add_library(companion_proxy companion/ProxyServer/ProxyServer.cpp companion/LoopbackPolicy/LoopbackPolicy.cpp)
target_link_libraries(companion_proxy PUBLIC companion_upstream)
add_library(companion_assistant assistant/LocalAssistantService/LocalAssistantService.cpp storage/CredentialVault/CredentialVault.cpp)
target_sources(companion_assistant PRIVATE assistant/PrivateMigration/PrivateMigration.cpp)
target_sources(companion_assistant PRIVATE assistant/PersonalSpeech/PersonalSpeech.cpp assistant/WindowsCloudSpeech/WindowsCloudSpeech.cpp
    ../backend/ai/speech/application/SpeechText/SpeechText.cpp
    ../backend/ai/speech/application/WaveCodec/WaveCodec.cpp
    ../backend/ai/speech/application/SpeechWire/SpeechWire.cpp
    ../backend/ai/speech/application/CloudSpeechPolicy/CloudSpeechPolicy.cpp)
target_link_libraries(companion_assistant PRIVATE winhttp)
target_link_libraries(companion_assistant PUBLIC companion_proxy companion_private_data)
add_executable(nebula_companion companion/CompanionMain/CompanionMain.cpp)
target_link_libraries(nebula_companion PRIVATE companion_assistant)
if(MINGW)
    target_link_options(nebula_companion PRIVATE -municode)
endif()
add_executable(companion_tls_probe ../tests/client/tls_probe.cpp)
target_link_libraries(companion_tls_probe PRIVATE companion_tls)
add_executable(companion_http_pool_probe ../tests/client/http_pool_probe.cpp)
target_link_libraries(companion_http_pool_probe PRIVATE companion_upstream)
if(BUILD_TESTING)
    add_executable(companion_speech_contract ../tests/function3/unit/speech_policy.cpp)
    target_link_libraries(companion_speech_contract PRIVATE companion_assistant)
    add_test(NAME companion_speech_contract COMMAND companion_speech_contract)
    add_executable(companion_action_contract ../tests/client/action_contract.cpp)
    target_link_libraries(companion_action_contract PRIVATE companion_private_data)
    add_test(NAME companion_action_contract COMMAND companion_action_contract)
    add_executable(companion_loopback_policy_contract ../tests/client/loopback_policy_contract.cpp)
    target_link_libraries(companion_loopback_policy_contract PRIVATE companion_proxy)
    add_test(NAME companion_loopback_policy_contract COMMAND companion_loopback_policy_contract)
    add_executable(companion_storage_contract ../tests/client/storage_contract.cpp)
    target_link_libraries(companion_storage_contract PRIVATE companion_storage)
    add_test(NAME companion_storage_contract COMMAND companion_storage_contract)
    add_executable(companion_private_data_contract ../tests/client/private_data_contract.cpp)
    target_link_libraries(companion_private_data_contract PRIVATE companion_private_data)
    add_test(NAME companion_private_data_contract COMMAND companion_private_data_contract)
endif()
