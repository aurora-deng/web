cmake_minimum_required(VERSION 3.16)

set(NEBULA_SYSTEMS_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(nebula_systems LANGUAGES CXX)
    include(CTest)
    set(NEBULA_SYSTEMS_STANDALONE ON)
endif()

option(NEBULA_ENABLE_FLEET "Build the C++ fleet coordinator and SQLite ledgers" ${NEBULA_SYSTEMS_STANDALONE})
option(NEBULA_FLEET_GRPC "Build the fleet gRPC executables using an installed SDK" OFF)
option(NEBULA_FLEET_NAV2 "Build the fleet edge's optional ROS 2 Nav2 simulation adapter" OFF)
if(NEBULA_FLEET_GRPC AND NOT NEBULA_ENABLE_FLEET)
    message(FATAL_ERROR "NEBULA_FLEET_GRPC requires NEBULA_ENABLE_FLEET")
endif()

find_package(Threads REQUIRED)
add_subdirectory(../workflow workflow)
if(NEBULA_ENABLE_FLEET)
    add_subdirectory(../fleet fleet)
endif()
if(NEBULA_FLEET_NAV2)
    if(NOT NEBULA_FLEET_GRPC OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Nav2 simulation requires Linux, NEBULA_FLEET_GRPC and a sourced ROS 2 environment")
    endif()
    add_subdirectory(../../robotics/nav2_gateway nav2_gateway)
    target_compile_definitions(nebula_fleet_edge PRIVATE NEBULA_FLEET_NAV2=1)
    target_link_libraries(nebula_fleet_edge PRIVATE nebula_nav2_navigation)
endif()
if(BUILD_TESTING)
    add_subdirectory(../../tests/cpp_systems cpp_systems_tests)
endif()
option(NEBULA_FLEET_BENCHMARK "Build the isolated Fleet ACK benchmark" OFF)
if(NEBULA_FLEET_BENCHMARK)
    if(NOT NEBULA_FLEET_GRPC)
        message(FATAL_ERROR "The Fleet ACK benchmark requires NEBULA_FLEET_GRPC")
    endif()
    add_subdirectory(../../benchmarks/cpp_systems cpp_systems_benchmarks)
endif()
